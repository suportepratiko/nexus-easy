1. Estratégia OTC

Baseada em candles em posições fixas (-1, -3, -5, -9). Só usa open e close.

def analyze(candles, asset=None):

"""

Estratégia OTC baseada em candles específicos (-1, -3, -5, -9).

Regras:

- CALL: c0 > c2 e c2 > o2 e c4 > c8

- PUT: c0 < c2 e c2 < o2 e c4 < c8

"""

c_list = list(candles)

if len(c_list) < 9:

return None

closes = [float(c["close"]) for c in c_list]

opens = [float(c["open"]) for c in c_list]

c0 = closes[-1]

c2 = closes[-3]

o2 = opens[-3]

c4 = closes[-5]

c8 = closes[-9]

if (c0 > c2) and (c2 > o2) and (c4 > c8):

return "call"

if (c0 < c2) and (c2 < o2) and (c4 < c8):

return "put"

return None



2. Estratégia SuperTrend

SuperTrend com ATR (period=18, multiplier=2.0). Usa open, close, max (high) e min (low). CALL na virada para alta, PUT na virada para baixa.

def analyze(candles, asset=None):

"""

SuperTrend (period=18, multiplier=2.0):

- CALL quando a direção muda para alta.

- PUT quando a direção muda para baixa.

"""

c_list = list(candles)

period = 18

multiplier = 2.0

if len(c_list) < period + 2:

return None

closes = [float(c["close"]) for c in c_list]

highs = [float(c["max"]) for c in c_list]

lows = [float(c["min"]) for c in c_list]

# True Range

tr = []

for i in range(len(c_list)):

if i == 0:

tr.append(highs[i] - lows[i])

continue

tr.append(max(

highs[i] - lows[i],

abs(highs[i] - closes[i - 1]),

abs(lows[i] - closes[i - 1]),

))

# ATR (EWMA com alpha=1/period), depois multiplicador

alpha = 1.0 / period

atr = [tr[0]]

for i in range(1, len(tr)):

atr.append(alpha * tr[i] + (1.0 - alpha) * atr[i - 1])

atr = [a * multiplier for a in atr]

supertrend = [None] * len(c_list)

direction = [0] * len(c_list)

for i in range(1, len(c_list)):

prev_st = supertrend[i - 1] if supertrend[i - 1] is not None else 0.0

prev_close = closes[i - 1]

curr_close = closes[i]

curr_h = highs[i]

curr_l = lows[i]

curr_atr = atr[i]

if curr_close > prev_st and prev_close > prev_st:

st = max(prev_st, curr_h - curr_atr)

elif curr_close < prev_st and prev_close < prev_st:

st = min(prev_st, curr_l + curr_atr)

elif curr_close > prev_st:

st = curr_h - curr_atr

else:

st = curr_l + curr_atr

supertrend[i] = st

if prev_close < prev_st and curr_close > prev_st:

direction[i] = 1

elif prev_close > prev_st and curr_close < prev_st:

direction[i] = -1

else:

direction[i] = direction[i - 1]

if len(direction) < 2:

return None

long_signal = direction[-2] != direction[-1] and direction[-1] == 1

short_signal = direction[-2] != direction[-1] and direction[-1] == -1

if long_signal:

return "call"

if short_signal:

return "put"

return None



Formato dos candles: cada item de candles é um dict com open, close, max (high), min (low) e from (timestamp).

Contrato: uma única função analyze(candles, asset=None) que retorna "call", "put" ou None.

Funções auxiliares disponíveis no ambiente das estratégias custom: _rsi(closes, period), _ema(values, period), _sma(values, period), _macd(closes), _atr(highs, lows, closes, period), _adx(highs, lows, closes, period).
